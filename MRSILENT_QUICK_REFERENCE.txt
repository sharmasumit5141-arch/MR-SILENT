━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

import "bot" as bot
import "user" as user
import "ai" as ai

fn start_command() {
    bot.reply("Hii {first_name}! Main MRSILENT se bana bot hu 🔥")
}

fn balance_command() {
    bal = user.get("balance") ?? 0
    bot.reply("Your balance: {bal}")
}

fn chat_command() {
    reply = ai.ask(text)
    bot.reply(reply)
}

fn ban_command() {
    is_group && bot.isAdmin(user_id) ? bot.ban(user_id) : bot.reply("Admins only!")
}
